event CalcularDescuento( as_tipo String, ac_precio )

dec{3} lc_Descuento

if as_tipo = 'ESTUDIANTE' THEN
	lc_Descuento = ac_precio*0.8
elseif as_tipo = 'SENIOR' THEN
	lc_Descuento = ac_precio*0.7
elseif as_tipo = 'VIP' THEN
	lc_Descuento = ac_precio*0.5
end if

return lc_Descuento
